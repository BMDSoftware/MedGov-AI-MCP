                                │

  └─────────────────────────────────────────────────────────────┘                                                                                                                             
                                │                                                                                                                                                             
                                ▼                                                                                                                                                             
  ┌─────────────────────────────────────────────────────────────┐                                                                                                                             
  │ 3. Notify callbacks: "before"                              │

                                │                                                                                                                                                             
                                ▼

  │                                                             │                                                                                                                             
  │    POST http://localhost:8000/mcp/tools/call               │                                                                                                                              
  │    {                                                        │                                                                                                                             
  │      "jsonrpc": "2.0",                                     │                                                                                                                              
  │      "method": "tools/call",                               │                                                                                                                              
  │      "params": {                                           │                                                                                                                              
  │        "name": "search",                                   │                                                                                                                              
  │        "arguments": {"type": "Patient"}                    │                                                                                                                              
  │      },                                                     │                                                                                                                             
  │      "id": 1                                               │                                                                                                                              
  │    }                                                        │                                                                                                                             
  └─────────────────────────────────────────────────────────────┘                                                                                                                             
                                │
